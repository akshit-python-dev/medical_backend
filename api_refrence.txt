# ============================================================================
# AUTHENTICATION ENDPOINTS
# ============================================================================

# Get JWT Token
# POST /api/token/
# Body: {"username": "admin", "password": "admin123"}
# Returns: {"access": "token_here", "refresh": "token_here"}

# Refresh Token
# POST /api/token/refresh/
# Body: {"refresh": "token_here"}
# Returns: {"access": "new_token_here"}


# ============================================================================
# PATIENT ENDPOINTS
# ============================================================================

# List All Patients
# GET /api/patients/
# Query params: ?search=john&ordering=-created_at&page=1

# Create Patient
# POST /api/patients/
# Body: {
#   "first_name": "John",
#   "last_name": "Doe",
#   "email": "john@example.com",
#   "phone": "555-1234",
#   "date_of_birth": "1990-01-01",
#   "gender": "M",
#   "address": "123 Main St",
#   "medical_history": "Asthma"
# }

# Get Patient Details
# GET /api/patients/1/

# Update Patient
# PATCH /api/patients/1/
# Body: {"first_name": "Jane"}

# Delete Patient
# DELETE /api/patients/1/

# Get Patient Medical History
# GET /api/patients/1/medical_history/

# Get Patient Upcoming Appointments
# GET /api/patients/1/upcoming_appointments/

# Get Patient Billing Summary
# GET /api/patients/1/billing_summary/

# ============================================================================
# FILTERING & SEARCH EXAMPLES
# ============================================================================

# Search by name
# GET /api/patients/?search=john

# Search and filter by status
# GET /api/appointments/?search=john&status=completed

# Sort by date (newest first)
# GET /api/patients/?ordering=-created_at

# Sort by date (oldest first)
# GET /api/patients/?ordering=created_at

# Pagination (page 2, 20 items per page)
# GET /api/patients/?page=2

# Combine filters and search
# GET /api/appointments/?search=john&status=scheduled&ordering=-appointment_date&page=1


# ============================================================================
# HEADERS FOR ALL REQUESTS
# ============================================================================

# Required Headers:
# Authorization: Bearer YOUR_ACCESS_TOKEN
# Content-Type: application/json
#
# Example:
# curl -X GET http://localhost:8000/api/patients/ \
#   -H "Authorization: Bearer eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9..." \
#   -H "Content-Type: application/json"


# ============================================================================
# RESPONSE STATUS CODES
# ============================================================================

# 200 OK               - Request successful, data returned
# 201 Created          - Resource created successfully
# 204 No Content       - Resource deleted successfully
# 400 Bad Request      - Invalid input data
# 401 Unauthorized     - Missing or invalid token
# 403 Forbidden        - Permission denied
# 404 Not Found        - Resource doesn't exist
# 409 Conflict         - Duplicate resource (e.g., duplicate email)
# 500 Server Error     - Server problem


# ============================================================================
# COMMON RESPONSE FORMATS
# ============================================================================

# Success Response (Single Object):
# {
#   "id": 1,
#   "first_name": "John",
#   "last_name": "Doe",
#   "email": "john@example.com",
#   ...
# }

# Success Response (List):
# {
#   "count": 100,
#   "next": "http://localhost:8000/api/patients/?page=2",
#   "previous": null,
#   "results": [
#     {
#       "id": 1,
#       "first_name": "John",
#       ...
#     }
#   ]
# }

# Error Response:
# {
#   "field_name": ["Error message"],
#   "another_field": ["Another error message"]
# }

# Auth Error Response:
# {
#   "detail": "Authentication credentials were not provided."
# }
